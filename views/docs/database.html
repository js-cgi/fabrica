Database

Query builder for SQLite

Connection

The database is connected automatically in index.js. Use DB in your controllers:

import { DB } from "../../framework/DB.js";

Query Builder

Select

// Get all rows
const users = DB.table("users").get();

// With conditions
const admins = DB.table("users")
    .where("role", "admin")
    .orderBy("name")
    .get();

// Single row
const user = DB.table("users").where("id", 1).first();

// Select specific columns
const names = DB.table("users").select("id", "name").get();

// Count
const total = DB.table("users").where("active", 1).count();

// Operators
DB.table("users").where("age", ">", 18).get();

// Pagination
DB.table("posts").orderBy("created_at", "DESC").limit(10).offset(20).get();

Insert

const id = DB.table("users").insert({
    name: "Alice",
    email: "alice@example.com",
    password: Auth.hashPassword("secret")
});

Update

DB.table("users").where("id", 1).update({
    name: "Alice Smith"
});

Delete

DB.table("users").where("id", 1).delete();

Raw Queries

const results = DB.raw("SELECT * FROM users WHERE email LIKE ?", ["%@example.com"]);
DB.exec("DROP TABLE IF EXISTS temp_data");

Migrations

Create migration files in database/migrations/:

import { DB } from "../../framework/DB.js";

export function up() {
    DB.exec(`CREATE TABLE IF NOT EXISTS posts (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        title TEXT NOT NULL,
        body TEXT,
        user_id INTEGER,
        created_at TEXT NOT NULL
    )`);
}

Import and run them in index.js:

import { up as migration002 } from "./database/migrations/002_create_posts.js";
migration002();
